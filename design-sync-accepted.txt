# Различия между макетом Claude Design (проект «Classical», apps/marginalia/app.css)
# и собранными стилями приложения, принятые осознанно.
#
# Сверка:
#   python3 ~/.claude/skills/skill-design-sync/scripts/css_diff.py \
#     <выгруженный app.css> dist/assets/index-*.css --accepted design-sync-accepted.txt
#
# Запись, переставшая соответствовать реальному расхождению, роняет сверку как
# устаревшая — список не может пережить различие, которое он описывает.

# Действия рецензии в шапке были текстовыми кнопками .btn-ghost и требовали
# ширины подписи, поэтому макет прячет их при ширине до 1180px. С c86e46a они
# иконочные, места почти не занимают и остаются доступными на узком экране.
# Причина записана в src/styles.css рядом с медиазапросом.
missing: @media (width<=1180px) || .header-actions .btn-ghost

# Класс иконочных кнопок экспорта, появившийся тем же изменением; в макете его
# нет, потому что там кнопки ещё текстовые.
extra: .action-button

# Приложение получило собственное хранилище, импорт рецензии и адаптивную
# раскладку. Соответствующих элементов в макете нет: он описывает вид версии,
# которая жила только в памяти вкладки и рассчитывалась на настольный экран.

# Рисованные значки для кнопок «Новая версия», «Удалить документ» и «Открыть
# рецензию». Шрифтовые не подошли: локальное подмножество Material Symbols
# содержит только девять глифов, а тянуть шрифт с чужого сервера ради трёх
# кнопок значило бы отправлять адрес читателя наружу при каждом открытии.
extra: .glyph

# Полоса о том, что открытая рецензия написана к другой версии статьи.
extra: .import-notice

# Отметка автосохранения: человеку, привыкшему нажимать «Сохранить», нужно
# видеть, что теперь это делается за него.
extra: .save-state
extra: .save-state[data-state=failed]

# Планшет в альбомной ориентации: оглавление уходит, две панели остаются.
extra: @media (width<=1040px) || :root
extra: @media (width<=1040px) || .workspace
extra: @media (width<=1040px) || .toc-bar
extra: @media (width<=1040px) || .pane-resizer[data-resize=toc]
extra: @media (width<=1040px) || .search-box

# Планшет в книжной ориентации: колонки не помещаются рядом, рецензия встаёт
# над документом, ручки изменения ширины скрыты.
extra: @media (width<=860px) || .workspace
extra: @media (width<=860px) || .pane-resizer
extra: @media (width<=860px) || .review-pane
extra: @media (width<=860px) || .brand span:not(.brand-mark)
extra: @media (width<=860px) || .document-controls
extra: @media (width<=860px) || .document-select
extra: @media (width<=860px) || .search-box

# Нижняя граница раскладки опущена с настольной до планшетной: ниже 980px макет
# полагался на горизонтальную прокрутку, а приложение перестраивает колонки.
differs: .app-shell -> min-width

# Макет берёт Cormorant Garamond и Lora у Google по @import. Приложение держит
# их подмножества рядом с собой: иначе Google получал бы адрес каждого читателя
# при открытии, а без сети заголовки теряли бы начертание. Отсюда исчезнувший
# @import и тринадцать блоков @font-face вместо одного иконочного.
differs: @import
differs: @font-face -> количество
differs: @font-face[0] -> font-display,font-family,src,unicode-range

# Полоса о том, где живёт рецензия, и кнопка установки приложения. В макете их
# нет: там нечего хранить, а устойчивость браузерного хранилища выдаётся только
# установленному приложению — это поведение среды, а не решение оформления.
extra: .storage-notice
extra: .storage-notice[hidden]
extra: .storage-notice p
extra: .storage-notice .btn

# Номера строк собраны в одну колонку слева от всего текста. В макете номер
# висел на левом краю своей строки и потому ложился на полосу цитаты и на
# маркер списка; поправку на отступ строки приносит --line-indent, а левое поле
# документа расширено под саму колонку.
differs: .source-line.line-origin::before -> right
differs: .document-body -> padding

# Прокрутка блока кода обрезала бы номер строки, ушедший левее этого блока.
# Замечание привязано к строке, поэтому видимый номер важнее точного вида
# длинной строки — она переносится.
differs: .document-body pre -> overflow
differs: .document-body pre -> overflow-x
extra: .document-body pre code

# Замечание с привязкой стало можно оставлять без слов: тип и процитированные
# строки уже высказывание. Надпись «Напишите комментарий» вместе со своим
# классом исчезла — сообщать больше не о чем.
missing: .draft-error

# Окна ручной вставки текста и переименования статьи. В макете их нет: там
# документ приходил только файлом и назывался только именем этого файла.
extra: .text-dialog
extra: .text-dialog::backdrop
extra: .text-dialog .input
extra: .text-dialog textarea.input
extra: .dialog-hint
extra: .dialog-actions
