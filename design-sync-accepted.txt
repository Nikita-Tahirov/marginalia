# Действия рецензии в шапке были текстовыми кнопками .btn-ghost и требовали
# ширины подписи, поэтому макет прячет их при ширине до 1180px. С c86e46a они
# иконочные, места почти не занимают и остаются доступными на узком экране.
# Причина записана в src/styles.css рядом с медиазапросом.
missing: @media (width<=1180px) || .header-actions .btn-ghost

# Класс иконочных кнопок экспорта, появившийся тем же изменением; в макете его
# нет, потому что там кнопки ещё текстовые.
extra: .action-button
